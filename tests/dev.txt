<nav
  id="menu"
  class="sidebar-menu mt-0 mb-2 br-0 bt-0"
  data-relapse="sidebar">

  {% if page.url contains '/rules/' %}
      <ul
      class="list rel my-0"
   >

      </ul>
  {% endif %}

  {%
    liquid
    assign sidebar = navigation | navigate
  %}

  {{ sidebar.rules }}

  {% for parent in sidebar %}

    {% liquid
      #
      # PARENT NAVIGATION GROUP
      #
      assign mapped_groups = parent.links | map: 'url'
      assign parent_expand = false
      if mapped_groups contains page.url
        assign parent_expand = true
      endif
    %}

    {% if parent.url %}
      <a
        spx-scroll="0"
        href="{{ parent.url | url }}"
        {% if parent_expand %}
          aria-expanded="true"
          class="menu-link active"
        {% else %}
          class="menu-link"
        {% endif %}>
        {{ parent.title }}
      </a>
    {% else %}
      <button
        type="button"
        {% if parent_expand %}
          aria-expanded="true"
          class="menu-link active"
        {% else %}
          class="menu-link"
        {% endif %}>
        {{ parent.title }}
      </button>
    {% endif %}
    <ul
      class="list rel my-0"
      {% unless parent_expand %}
        hidden="true"
      {% endunless %}>
      {% for child in parent.links %}
        {%
          liquid
          #
          # CHILD NAVIGATION GROUP
          #
          assign mapped_child = child.links | map: 'url'
          assign child_expand = false
          if mapped_child contains page.url
            assign child_expand = true
          endif
        %}
        {% if child.links %}
          <li data-relapse>
            <button
              type="button"
              {% if page.url == child.url or child_expand %}
                aria-expanded="true"
                class="menu-link active"
              {% else %}
                class="child-link"
              {% endif %}>
              {{ child.title }}
            </button>
            <ul
              class="child-list rel pl-2"
              {% unless child_expand %}
                hidden="true"
              {% endunless %}>
              {% for child in child.links %}
                <li class="pl-1">
                  <a
                    spx-scroll="0"
                    href="{{ child.url | url }}"
                    {% if page.url == child.url %}
                      class="active"
                    {% endif %}>
                    {{ child.title }}
                  </a>
                </li>
              {% endfor %}
            </ul>
          </li>
        {% else %}
          <li>
            <a
              spx-scroll="0"
              href="{{ child.url | url }}"
              aria-disabled="true"
              {% if page.url == child.url %}
                class="active"
              {% endif %}>
              {{ child.title }}
            </a>
          </li>
        {% endif %}
      {% endfor %}
    </ul>
  {% endfor %}
</nav>
